#include <ESP32Servo.h>

// ================= PIN DEFINITIONS =================

const int BUTTON_PIN = 23;
const int SERVO_PIN  = 18;

const int LED1_PIN = 33;
const int LED2_PIN = 32;
const int LED3_PIN = 26;
const int LED4_PIN = 25;

const int BUZZER_PIN = 21;

// ================= SETTINGS =================

const unsigned long FEED_INTERVAL = 20000; // 20 sec for testing

const int SERVO_IDLE = 0;
const int SERVO_FEED = 60;

const unsigned long SERVO_MOVE_DELAY = 500;
const unsigned long SERVO_HOLD_TIME  = 10000;

// ================= OBJECTS =================

Servo feederServo;

unsigned long lastFeedTime = 0;
unsigned long lastWarning = 0;

// ================= FUNCTIONS =================

void feedNow();
void blinkAllLEDs(unsigned long duration);
void lowFoodWarning();

// ================= SETUP =================

void setup() {

  Serial.begin(115200);

  pinMode(BUTTON_PIN, INPUT_PULLUP);

  pinMode(LED1_PIN, OUTPUT);
  pinMode(LED2_PIN, OUTPUT);
  pinMode(LED3_PIN, OUTPUT);
  pinMode(LED4_PIN, OUTPUT);

  pinMode(BUZZER_PIN, OUTPUT);

  digitalWrite(LED1_PIN, LOW);
  digitalWrite(LED2_PIN, LOW);
  digitalWrite(LED3_PIN, LOW);
  digitalWrite(LED4_PIN, LOW);

  digitalWrite(BUZZER_PIN, LOW);

  feederServo.setPeriodHertz(50);
  feederServo.attach(SERVO_PIN, 500, 2400);
  feederServo.write(SERVO_IDLE);

  lastFeedTime = millis();

  Serial.println("ESP32 Pet Feeder Ready");
}

// ================= LOOP =================

void loop() {

  unsigned long now = millis();

  // -------- Manual Feed --------

  if (digitalRead(BUTTON_PIN) == LOW) {

    Serial.println("Manual Feed Triggered");

    feedNow();

    lastFeedTime = millis();

    delay(500); // prevent multiple presses
  }

  // -------- Automatic Feed --------

  if (now - lastFeedTime >= FEED_INTERVAL) {

    Serial.println("Automatic Feed Triggered");

    feedNow();

    lastFeedTime = millis();
  }

  // -------- Low Food Warning --------

  if (now - lastWarning > 5000) {

    lowFoodWarning();

    lastWarning = now;
  }

}

// ================= FEED FUNCTION =================

void feedNow() {

  Serial.println("Feeding Started");

  digitalWrite(BUZZER_PIN, HIGH);

  feederServo.write(SERVO_FEED);

  blinkAllLEDs(SERVO_MOVE_DELAY);

  blinkAllLEDs(SERVO_HOLD_TIME);

  feederServo.write(SERVO_IDLE);

  blinkAllLEDs(SERVO_MOVE_DELAY);

  digitalWrite(BUZZER_PIN, LOW);

  Serial.println("Feeding Completed");
}

// ================= LED BLINK =================

void blinkAllLEDs(unsigned long duration) {

  unsigned long start = millis();
  bool state = false;

  while (millis() - start < duration) {

    state = !state;

    digitalWrite(LED1_PIN, state);
    digitalWrite(LED2_PIN, state);
    digitalWrite(LED3_PIN, state);
    digitalWrite(LED4_PIN, state);

    delay(150);
  }

}

// ================= LOW FOOD WARNING =================

void lowFoodWarning() {

  digitalWrite(LED1_PIN, HIGH);
  digitalWrite(BUZZER_PIN, HIGH);

  delay(120);

  digitalWrite(LED1_PIN, LOW);
  digitalWrite(BUZZER_PIN, LOW);

}
